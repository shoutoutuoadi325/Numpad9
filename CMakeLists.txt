cmake_minimum_required(VERSION 3.20)
project(Numpad9 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(NUMPAD9_BUILD_APP "Build the Windows onboarding app" ON)
include(CTest)
add_library(numpad9_core STATIC src/core/engine.cpp)
target_include_directories(numpad9_core PUBLIC src src/core)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive- /EHsc)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE numpad9_core)
  add_test(NAME core COMMAND core_tests "${CMAKE_CURRENT_SOURCE_DIR}/data/lexicon.tsv")
endif()
if(WIN32)
  enable_language(RC)
  # TSF is accessed through COM; its GUIDs are supplied by uuid.lib.
  add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  file(GLOB IME_SOURCES CONFIGURE_DEPENDS src/ime/*.cpp)
  add_library(Numpad9Ime SHARED ${IME_SOURCES} src/ime/ime.def src/app/ime.rc)
  target_include_directories(Numpad9Ime PRIVATE src/app)
  target_link_libraries(Numpad9Ime PRIVATE numpad9_core ole32 oleaut32 advapi32 user32 gdi32 uuid uiautomationcore)
  set_target_properties(Numpad9Ime PROPERTIES PREFIX "")
  if(NUMPAD9_BUILD_APP)
    add_executable(Numpad9 WIN32 src/app/main.cpp src/app/app.rc)
    target_include_directories(Numpad9 PRIVATE src/ime src/app)
    target_link_libraries(Numpad9 PRIVATE comctl32 ole32 oleaut32 shell32 user32 gdi32 uuid)
    if(MINGW)
      target_link_options(Numpad9 PRIVATE -municode)
    endif()
  endif()
  if(MINGW)
    target_link_options(Numpad9Ime PRIVATE -static -static-libgcc -static-libstdc++)
    if(NUMPAD9_BUILD_APP)
      target_link_options(Numpad9 PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
  endif()
  add_custom_command(TARGET Numpad9Ime POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/data/lexicon.tsv" "$<TARGET_FILE_DIR:Numpad9Ime>/lexicon.tsv")
  if(BUILD_TESTING)
    add_executable(ime_smoke tests/ime_smoke.cpp)
    target_include_directories(ime_smoke PRIVATE src/ime)
    target_link_libraries(ime_smoke PRIVATE ole32 uuid)
    if(MINGW)
      target_link_options(ime_smoke PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME ime_com_lifecycle COMMAND ime_smoke "$<TARGET_FILE:Numpad9Ime>")
  endif()
  install(TARGETS Numpad9Ime RUNTIME DESTINATION . LIBRARY DESTINATION .)
  if(NUMPAD9_BUILD_APP)
    install(TARGETS Numpad9 RUNTIME DESTINATION .)
  endif()
  install(FILES data/lexicon.tsv DESTINATION .)
  install(FILES data/NOTICE.txt data/SOURCES.json LICENSE DESTINATION .)
  install(DIRECTORY data/licenses DESTINATION .)
  install(FILES docs/dictionary.md DESTINATION .)
endif()
